-- creating sales flat

CREATE TABLE sales_flat (
    Date DATE,
    InvoiceNo VARCHAR(50),
    ProductCode VARCHAR(255),
    ProductName VARCHAR(255),
    Qty DECIMAL(10,2),
    Discount DECIMAL(5,2),
    in_sales DECIMAL(12,2),
    Brand VARCHAR(255),
    CustomerCategory VARCHAR(255),
    ProductCategory VARCHAR(255),
    Route VARCHAR(255),
    RepId INTEGER,
    RepCode VARCHAR(255),
    RepName VARCHAR(255),
    CustomerCode VARCHAR(100),
    CustomerName VARCHAR(100),
    ASM VARCHAR(100),
    RSM VARCHAR(100),
    Distributor VARCHAR(255),
    Type VARCHAR(10)
);

-- Populate sales_flat

INSERT INTO sales_flat (
    Date, InvoiceNo, ProductCode, ProductName, Qty, Discount, in_sales, Brand,
    CustomerCategory, ProductCategory, Route, RepId, RepCode, RepName, CustomerCode,CustomerName,
    ASM, RSM, Distributor, Type
)
SELECT 
    i.Date,
    i.SerialNo AS InvoiceNo,
    p.Code AS ProductCode,
    p.Name AS ProductName,
    il.Qty,
    il.DiscountPct AS Discount,
    ROUND((il.Qty * il.Price * (1 - il.DiscountPct / 100)), 2) AS in_sales,
    b.Name AS Brand,
    cusCat.Code AS CustomerCategory,
    pc.Name AS ProductCategory,
    r.Name AS Route,
    shn.Id as RepId,
    shn.Code as RepCode,
    shn.Name AS RepName,
    ep.Code AS CustomerCode,
    ep.Name AS CustomerName,
    'ASM' AS ASM,         -- Placeholder, you can join later if needed
    'RSM' AS RSM,         -- Placeholder
    dis.Name AS Distributor,
    CASE 
        WHEN i.TypeId = 103 THEN 'sales' 
        ELSE 'return' 
    END AS Type
FROM invoices i
INNER JOIN invoice_lines il ON i.Id = il.InvoiceId
INNER JOIN products p ON p.Id = il.ProductId
INNER JOIN product_categories pc ON pc.Id = p.CategoryId
INNER JOIN brands b ON b.Id = p.BrandId
INNER JOIN external_parties ep ON ep.Id = i.OtherPartyId
LEFT JOIN external_party_categories cusCat ON cusCat.Id = ep.CategoryId
LEFT JOIN routes r ON r.Id = i.RouteId
INNER JOIN sales_hierarchy_nodes shn ON shn.Id = i.SalesHierarchyNodeId
INNER JOIN external_parties dis ON dis.Id = i.BranchId
WHERE i.TypeId IN (103, 107) AND i.Deleted = false;
